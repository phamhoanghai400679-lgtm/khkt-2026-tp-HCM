const form = document.getElementById("uploadForm"); form.addEventListener("submit", async function(e) { e.preventDefault(); const formData = new FormData(form); const response = await fetch("/predict", { method: "POST", body: formData }); const data = await response.json(); document.getElementById("light").textContent = data["Nhẹ"] || "--"; document.getElementById("medium").textContent = data["Trung bình"] || "--"; document.getElementById("severe").textContent = data["Nặng"] || "--"; document.getElementById("conclusion").textContent = data["Kết luận"] || "Không có kết quả"; document.getElementById("resultTable").style.display = "table"; });
